//@version=6 indicator("DM FX - ZigZag Swing Signals", overlay=true, max_lines_count=500, max_labels_count=500, max_boxes_count=200) // ===== SETTINGS ===== pivotLen = input.int(5, "Swing Sensitivity", minval=2, maxval=30) atrLen = input.int(14, "ATR Period", minval=1) zoneATR = input.float(0.35, "Zone Thickness (ATR)", minval=0.05, step=0.05) slATR = input.float(0.5, "Stop Loss Buffer (ATR)", minval=0.1, step=0.1) extendBars = input.int(25, "Zone Extension", minval=5, maxval=200) showZones = input.bool(true, "Show Red/Green Zones") showLabels = input.bool(true, "Show BUY/SELL Labels") showStops = input.bool(true, "Show Stop Loss Lines") // ===== CALCULATIONS ===== atr = ta.atr(atrLen) pivotHigh = ta.pivothigh(high, pivotLen, pivotLen) pivotLow = ta.pivotlow(low, pivotLen, pivotLen) var int lastX = na var float lastY = na var int lastType = 0 bool buySignal = not na(pivotLow) bool sellSignal = not na(pivotHigh) // ===== SWING HIGH / SELL ===== if sellSignal int x = bar_index - pivotLen float y = pivotHigh float swingATR = atr[pivotLen] if not na(lastX) and lastType != 1 line.new(lastX, lastY, x, y, color=color.black, width=2) if showZones and not na(swingATR) box.new(left=x, top=y + swingATR * zoneATR, right=x + extendBars, bottom=y - swingATR * zoneATR, bgcolor=color.new(color.red, 85), border_color=color.new(color.red, 45)) if showLabels label.new(x, y, "SELL\nSwing High", style=label.style_label_down, color=color.red, textcolor=color.white, size=size.small) if showStops and not na(swingATR) line.new(x, y + swingATR * slATR, x + extendBars, y + swingATR * slATR, color=color.red, style=line.style_dashed) lastX := x lastY := y lastType := 1 // ===== SWING LOW / BUY ===== if buySignal int x = bar_index - pivotLen float y = pivotLow float swingATR = atr[pivotLen] if not na(lastX) and lastType != -1 line.new(lastX, lastY, x, y, color=color.black, width=2) if showZones and not na(swingATR) box.new(left=x, top=y + swingATR * zoneATR, right=x + extendBars, bottom=y - swingATR * zoneATR, bgcolor=color.new(color.lime, 85), border_color=color.new(color.green, 45)) if showLabels label.new(x, y, "BUY\nSwing Low", style=label.style_label_up, color=color.green, textcolor=color.white, size=size.small) if showStops and not na(swingATR) line.new(x, y - swingATR * slATR, x + extendBars, y - swingATR * slATR, color=color.green, style=line.style_dashed) lastX := x lastY := y lastType := -1 // ===== ALERTS ===== alertcondition(buySignal, title="BUY Signal", message="DM FX: Confirmed swing-low BUY signal on {{ticker}} at {{close}}") alertcondition(sellSignal, title="SELL Signal", message="DM FX: Confirmed swing-high SELL signal on {{ticker}} at {{close}}")
